type Policy = 'reader' | 'naive' | 'greedy' | 'defensive'
const eff = (def: CardDef, kind: string, target?: string) => def.effects.filter((e: any) => e.kind === kind && (target ? e.target === target : true)).reduce((s, e: any) => s + (e.amount ?? 0), 0)
const dmgOf = (x: CardInstance) => eff(getCardDef(x.defId), 'damage', 'enemy')
const blockOf = (x: CardInstance) => eff(getCardDef(x.defId), 'block')
const healOf = (x: CardInstance) => eff(getCardDef(x.defId), 'heal')
const selfDmgOf = (x: CardInstance) => eff(getCardDef(x.defId), 'damage', 'self')
const debuffOf = (x: CardInstance) => getCardDef(x.defId).effects.filter((e: any) => e.kind === 'debuff' && e.target === 'enemy').reduce((s, e: any) => s + e.amount, 0)
const costOf = (x: CardInstance) => getCardDef(x.defId).cost + (x.costModifier ?? 0)
const utilOf = (x: CardInstance) => getCardDef(x.defId).effects.reduce((s, e: any) => {
  if (e.kind === 'draw' || e.kind === 'gainAp') return s + e.amount * 4
  if (e.kind === 'resonance') return s + e.amount * 2.5
  if ((e.kind === 'buff' && e.target === 'self') || (e.kind === 'debuff' && e.target === 'enemy')) return s + e.amount * e.rounds
  return s
}, 0)
const incomingOf = (s: GameState) => (s.enemy.intent ? enemyActionTotal(s.enemy.intent) : 0)
const effIncoming = (s: GameState) => Math.max(0, incomingOf(s) + sumBuff(s.enemy.buffs, 'atk'))
const projectedHp = (s: GameState) => s.player.hp - Math.max(0, effIncoming(s) - s.player.block)
function bonusDmgIf(state: GameState, x: CardInstance): number {
  const def = getCardDef(x.defId)
  if (!def.bonus) return 0
  const extra = def.bonus.effects.filter((e: any) => e.kind === 'damage' && e.target === 'enemy').reduce((s, e: any) => s + e.amount, 0)
  if (extra === 0) return 0
  const inc = incomingOf(state)
  switch (def.bonus.when) {
    case 'combo': return state.cardsPlayedThisRound >= 1 ? extra : 0
    case 'charged': return state.resonance.value >= RULES.cardBonus.chargedThreshold ? extra : 0
    case 'enemyBig': return inc >= RULES.cardBonus.enemyBigThreshold ? extra : 0
    case 'lowHp': return state.player.hp <= Math.floor(state.player.maxHp * 0.5) ? extra : 0
    case 'blocked': return inc > 0 && state.player.block + blockOf(x) >= inc ? extra : 0
  }
  return 0
}
const totalDmg = (s: GameState, x: CardInstance) => dmgOf(x) + bonusDmgIf(s, x) + (dmgOf(x) > 0 ? Math.max(0, sumBuff(s.player.buffs, 'atk')) : 0)
function canKillNow(state: GameState, affordable: CardInstance[]): boolean {
  const dmgCards = affordable.filter((x) => dmgOf(x) > 0).sort((p, q) => totalDmg(state, q) / costOf(q) - totalDmg(state, p) / costOf(p))
  let ap = state.ap.current
  let sum = 0
  for (const x of dmgCards) if (costOf(x) <= ap) { ap -= costOf(x); sum += totalDmg(state, x) }
  return sum >= state.enemy.hp + state.enemy.block
}
function pickCard(state: GameState, policy: Policy): CardInstance | null {
  const affordable = state.hand.filter((x) => costOf(x) <= state.ap.current)
  if (affordable.length === 0) return null
  if (policy === 'naive') return affordable[0]
  const inc = effIncoming(state)
  const hpAfter = projectedHp(state)
  const lethal = hpAfter <= 0
  const dangerous = hpAfter < state.player.maxHp * 0.35
  const guarded = state.player.block >= inc
  const byDmg = (arr: CardInstance[]) => arr.filter((x) => dmgOf(x) > 0).sort((p, q) => totalDmg(state, q) / costOf(q) - totalDmg(state, p) / costOf(p))
  const bestBlock = () => affordable.filter((x) => blockOf(x) > 0).sort((p, q) => blockOf(q) - blockOf(p))[0]
  const bestDebuff = () => affordable.filter((x) => debuffOf(x) > 0).sort((p, q) => debuffOf(q) - debuffOf(p))[0]
  const bestHeal = () => affordable.filter((x) => healOf(x) > 0).sort((p, q) => healOf(q) - healOf(p))[0]
  const bestUtil = () => affordable.filter((x) => utilOf(x) > 0 && dmgOf(x) === 0 && blockOf(x) === 0 && healOf(x) === 0).sort((p, q) => utilOf(q) / costOf(q) - utilOf(p) / costOf(p))[0]
  if (policy === 'greedy') return byDmg(affordable)[0] ?? bestUtil() ?? bestBlock() ?? bestHeal() ?? affordable.sort((p, q) => costOf(q) - costOf(p))[0]
  if (policy === 'defensive') {
    if (!guarded) { const b = bestBlock(); if (b) return b }
    const h = bestHeal()
    if (h && state.player.hp < state.player.maxHp * 0.6) return h
    return byDmg(affordable)[0] ?? bestUtil() ?? affordable.sort((p, q) => costOf(q) - costOf(p))[0]
  }
  const free = affordable.find((x) => { const def = getCardDef(x.defId); return def.effects.some((e: any) => e.kind === 'gainAp' && e.amount >= def.cost) })
  if (free) return free
  if (canKillNow(state, affordable)) { const d = byDmg(affordable)[0]; if (d) return d }
  if ((lethal || dangerous) && !guarded) {
    const b = bestBlock(); if (b) return b
    const db = bestDebuff(); if (db) return db
    const h = bestHeal(); if (h) return h
  }
  if (lethal) { const h = bestHeal(); if (h) return h }
  const safe = affordable.filter((x) => selfDmgOf(x) === 0 || state.player.hp - selfDmgOf(x) - Math.max(0, inc - state.player.block) > 0)
  const d = byDmg(safe)[0]
  if (d) return d
  const u = bestUtil(); if (u) return u
  const h = bestHeal(); if (h && state.player.hp < state.player.maxHp * 0.7) return h
  const b = bestBlock(); if (b && !guarded && inc > 0) return b
  return null
}
function pickOracle(state: GameState, policy: Policy): number {
  if (state.divination.remaining <= 0 || state.divination.usedThisRound) return -1
  if (policy === 'naive' || policy === 'greedy') return 2
  if (policy === 'defensive') return 0
  const inc = effIncoming(state)
  const hpAfter = projectedHp(state)
  const lethal = hpAfter <= 0
  const dangerous = hpAfter < state.player.maxHp * 0.35
  if (state.enemy.hp + state.enemy.block <= 4) return 2
  if (lethal && inc > 0) return 0
  const roundsLeft = RULES.totalRounds - state.round + 1
  if (state.divination.remaining >= roundsLeft) return dangerous && inc > 0 ? 0 : 2
  if (dangerous && inc > 0) return 0
  if (roundsLeft <= state.divination.remaining) return inc >= 10 && state.player.block < inc ? 0 : 2
  return -1
}
type Res = { status: string; round: number; score: number; hp: number }
function play(godId: GodId, enemyId: EnemyId, diff: Difficulty, policy: Policy, seed: number, extra: { stake?: number; stakeChoice?: 'race' | 'pressure' | 'tempo'; daily?: boolean }): Res {
  const deck = getRecommendedDeck(godId)
  let { state } = applyAction(null, {
    type: 'START_GAME', seed: `d245r-${seed}`, godId, enemyId, deck, difficulty: diff,
    ...(extra.daily ? { mode: 'daily' as const, dailyKey: '2026-09-28', modifier: RULES.daily.modifier } : {}),
    ...(extra.stake ? { stake: extra.stake, ...(extra.stakeChoice ? { stakeChoice: extra.stakeChoice } : {}) } : {}),
  })
  let guard = 0
  while (state.status === 'playing' && guard < 600) {
    guard++
    const card = pickCard(state, policy)
    if (card) { state = applyAction(state, { type: 'PLAY_CARD', uid: card.uid }).state; continue }
    const o = pickOracle(state, policy)
    if (o >= 0) { state = applyAction(state, { type: 'USE_DIVINATION', choiceIndex: o }).state; continue }
    state = applyAction(state, { type: 'END_ROUND' }).state
  }
  return { status: state.status, round: state.round, score: getFinalScore(state.score, state.stake), hp: state.player.hp }
}
