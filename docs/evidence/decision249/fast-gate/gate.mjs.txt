  if (!(await clickSel(page, '[data-testid="home-start"]'))) await clickText(page, '神を選ぶ')
  await page.waitForTimeout(400)
  if (await clickText(page, '新しく始める')) await page.waitForTimeout(300)
  await clickText(page, sc.god); await page.waitForTimeout(300)
  await clickText(page, 'この構成で始める'); await page.waitForTimeout(300)
  await clickText(page, sc.enemy); await page.waitForTimeout(500)
  await clickText(page, 'この構成でバトル開始')

async function playOne(page, cardName) {
  const t0 = Date.now()
  const ok = await page.evaluate((n) => { const el = [...document.querySelectorAll('.hand .card-view')].find((c) => (c.querySelector('.card-view-name')?.textContent.trim() ?? '').replace(/^[^\p{L}\p{N}]+\s*/u, '') === n && !c.disabled); if (!el) return false; window.__lock = null; window.__t0 = performance.now(); const b = document.querySelector('.end-round-button'); const mo = new MutationObserver(() => { if (!b.disabled && window.__lock === null) { window.__lock = performance.now() - window.__t0; mo.disconnect() } }); mo.observe(b, { attributes: true }); el.click(); return true }, cardName)
  if (!ok) return null
  const samples = []
  let lockMs = null
  for (let i = 0; i < 24; i++) {
    const o = await observe(page)
    o.t = Date.now() - t0
    samples.push(o)
    if (lockMs === null && o.endDisabled === false && o.t > 50) lockMs = o.t
    await page.waitForTimeout(25)
  }
  // 入力ロック：end-round が再び有効になるまで（cast 280ms ＋ 描画）
  while (lockMs === null && Date.now() - t0 < 3000) { const o = await observe(page); if (o.endDisabled === false) lockMs = Date.now() - t0; await page.waitForTimeout(10) }
  await page.waitForTimeout(400)
  const lockInPage = await page.evaluate(() => window.__lock)
  const union = (k) => [...new Set(samples.flatMap((s) => s[k]))]
  return { lockMs: lockInPage ?? lockMs, god: union('god'), godAnim: union('godAnim'), enemyIdle: union('enemyIdle'), enemyAnim: union('enemyAnim'), enemyReact: samples.some((s) => s.enemyReact), otomo: union('otomo'), otomoAnim: union('otomoAnim'), dealtMax: Math.max(...samples.map((s) => s.dealt)), ap: union('ap') }
}
