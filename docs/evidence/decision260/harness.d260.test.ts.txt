type Policy = 'reader' | 'naive' | 'greedy' | 'planner' | 'randomRO'
const isUlt = (x: EnemyActionDef | null | undefined) => !!x && (x.kind === 'special' || (x.kind === 'multiAttack' && !!x.special))
const isEnemyDebuff = (e: Effect): e is Extract<Effect, { kind: 'debuff' }> => e.kind === 'debuff' && e.target === 'enemy'
const eff = (def: CardDef, kind: string, target?: string) => def.effects.filter((e: any) => e.kind === kind && (target ? e.target === target : true)).reduce((s, e: any) => s + (e.amount ?? 0), 0)
const dmgOf = (x: CardInstance) => eff(getCardDef(x.defId), 'damage', 'enemy')
const blockOf = (x: CardInstance) => eff(getCardDef(x.defId), 'block')
const healOf = (x: CardInstance) => eff(getCardDef(x.defId), 'heal')
const selfDmgOf = (x: CardInstance) => eff(getCardDef(x.defId), 'damage', 'self')
const debuffOf = (x: CardInstance) => getCardDef(x.defId).effects.filter(isEnemyDebuff).reduce((s, e) => s + e.amount, 0)
const debuffRoundsOf = (x: CardInstance) => getCardDef(x.defId).effects.filter(isEnemyDebuff).reduce((s, e) => Math.max(s, e.rounds), 0)
const costOf = (x: CardInstance) => getCardDef(x.defId).cost + (x.costModifier ?? 0)
const utilOf = (x: CardInstance) => getCardDef(x.defId).effects.reduce((s, e: any) => {
  if (e.kind === 'draw' || e.kind === 'gainAp') return s + e.amount * 4
  if (e.kind === 'resonance') return s + e.amount * 2.5
  if ((e.kind === 'buff' && e.target === 'self') || (e.kind === 'debuff' && e.target === 'enemy')) return s + e.amount * e.rounds
  return s
}, 0)
const incomingOf = (s: GameState) => (s.enemy.intent ? enemyActionTotal(s.enemy.intent) : 0)
const effIncoming = (s: GameState) => Math.max(0, incomingOf(s) + sumBuff(s.enemy.buffs, 'atk'))
const projectedHp = (s: GameState) => s.player.hp - Math.max(0, effIncoming(s) - s.player.block)
function bonusDmgIf(state: GameState, x: CardInstance): number {
  const def = getCardDef(x.defId); if (!def.bonus) return 0
  const extra = def.bonus.effects.filter((e: any) => e.kind === 'damage' && e.target === 'enemy').reduce((s, e: any) => s + e.amount, 0)
  if (extra === 0) return 0
  const inc = incomingOf(state)
  switch (def.bonus.when) {
    case 'combo': return state.cardsPlayedThisRound >= 1 ? extra : 0
    case 'charged': return state.resonance.value >= RULES.cardBonus.chargedThreshold ? extra : 0
    case 'enemyBig': return inc >= RULES.cardBonus.enemyBigThreshold ? extra : 0
    case 'lowHp': return state.player.hp <= Math.floor(state.player.maxHp * 0.5) ? extra : 0
    case 'blocked': return inc > 0 && state.player.block + blockOf(x) >= inc ? extra : 0
  }
  return 0
}
const totalDmg = (s: GameState, x: CardInstance) => dmgOf(x) + bonusDmgIf(s, x) + (dmgOf(x) > 0 ? Math.max(0, sumBuff(s.player.buffs, 'atk')) : 0)
function maxDmgNow(state: GameState, affordable: CardInstance[]): number {
  const dmgCards = affordable.filter((x) => dmgOf(x) > 0).sort((p, q) => totalDmg(state, q) / costOf(q) - totalDmg(state, p) / costOf(p))
  let ap = state.ap.current, sum = 0
  for (const x of dmgCards) if (costOf(x) <= ap) { ap -= costOf(x); sum += totalDmg(state, x) }
  return sum
}
const canKillNow = (state: GameState, affordable: CardInstance[]) => maxDmgNow(state, affordable) >= state.enemy.hp + state.enemy.block
function upcomingUltimates(state: GameState): number { const def = ENEMIES.find((e) => e.id === state.enemy.defId)!; let n = 0; for (let r = state.round + 1; r <= RULES.totalRounds; r++) if (isUlt(def.actions[r - 1])) n++; return n }
function futureIncoming(state: GameState, r: number): number {
  const def = ENEMIES.find((e) => e.id === state.enemy.defId)!
  const raw = enemyActionTotal(def.actions[r - 1]), rawNow = enemyActionTotal(def.actions[state.round - 1])
  const ratio = rawNow > 0 && state.enemy.intent ? incomingOf(state) / rawNow : 1
  const k = r - state.round
  const debuff = state.enemy.buffs.filter((b: any) => b.stat === 'atk' && b.amount < 0 && ((b.remainingRounds ?? b.rounds ?? 1) > k)).reduce((s: number, b: any) => s + b.amount, 0)
  return Math.max(0, Math.round(raw * ratio) + debuff)
}
function cardValue(state: GameState, x: CardInstance): number {
  const inc = effIncoming(state); const d = totalDmg(state, x)
  const b = Math.min(blockOf(x), Math.max(0, inc - state.player.block)); const h = Math.min(healOf(x), state.player.maxHp - state.player.hp)
  return d + b + h * 0.8 + utilOf(x) * 0.5
}
type Ctx = { unblockedNow: number; pressure: number; hpAfterNow: number }
function plannerCtx(state: GameState): Ctx {
  const inc = effIncoming(state); const unblockedNow = Math.max(0, inc - state.player.block); const hpAfterNow = state.player.hp - unblockedNow
  let future = 0; for (let r = state.round + 1; r <= Math.min(RULES.totalRounds, state.round + 2); r++) future += futureIncoming(state, r)
  const hpHorizon = hpAfterNow - future
  return { unblockedNow, pressure: hpAfterNow <= 0 ? 3 : hpHorizon <= 0 ? 1.6 : hpHorizon < state.player.maxHp * 0.35 ? 1.2 : 0.6, hpAfterNow }
}
function plannerValue(state: GameState, x: CardInstance, ctx: Ctx): { v: number; def: number } {
  const dmg = totalDmg(state, x); const killable = state.enemy.hp + state.enemy.block
  let v = Math.min(dmg, killable) + (dmg >= killable && dmg > 0 ? 15 : 0)
  let def = Math.min(blockOf(x), ctx.unblockedNow)
  const db = debuffOf(x), dr = debuffRoundsOf(x)
  if (db > 0) for (let k = 0; k < dr; k++) { const r = state.round + k; if (r > RULES.totalRounds) break; const target = k === 0 ? ctx.unblockedNow : futureIncoming(state, r); def += Math.min(db, Math.max(0, target)) * (k === 0 ? 1 : 0.9) }
  const h = Math.min(healOf(x), state.player.maxHp - state.player.hp); def += h * (ctx.hpAfterNow < state.player.maxHp * 0.5 ? 1.0 : 0.6)
  v += utilOf(x) * 0.5 - selfDmgOf(x) * ctx.pressure
  return { v: v + def * ctx.pressure, def }
}
function mulberry32(a: number) { return () => { a |= 0; a = (a + 0x6d2b79f5) | 0; let t = Math.imul(a ^ (a >>> 15), 1 | a); t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t; return ((t ^ (t >>> 14)) >>> 0) / 4294967296 } }
function pickReader(state: GameState, policy: Policy, affordable: CardInstance[]): CardInstance | null {
  if (policy === 'naive') return affordable[0]
  const inc = effIncoming(state), hpAfter = projectedHp(state), lethal = hpAfter <= 0, dangerous = hpAfter < state.player.maxHp * 0.35, guarded = state.player.block >= inc
  const byDmg = (arr: CardInstance[]) => arr.filter((x) => dmgOf(x) > 0).sort((p, q) => totalDmg(state, q) / costOf(q) - totalDmg(state, p) / costOf(p))
  const bestBlock = () => affordable.filter((x) => blockOf(x) > 0).sort((p, q) => blockOf(q) - blockOf(p))[0]
  const bestDebuff = () => affordable.filter((x) => debuffOf(x) > 0).sort((p, q) => debuffOf(q) * debuffRoundsOf(q) - debuffOf(p) * debuffRoundsOf(p))[0]
  const bestHeal = () => affordable.filter((x) => healOf(x) > 0).sort((p, q) => healOf(q) - healOf(p))[0]
  const bestUtil = () => affordable.filter((x) => utilOf(x) > 0 && dmgOf(x) === 0 && blockOf(x) === 0 && healOf(x) === 0).sort((p, q) => utilOf(q) / costOf(q) - utilOf(p) / costOf(p))[0]
  if (policy === 'greedy') { const d = byDmg(affordable)[0]; if (d) return d; return bestUtil() ?? bestBlock() ?? bestHeal() ?? affordable.sort((p, q) => costOf(q) - costOf(p))[0] }
  const free = affordable.find((x) => { const def = getCardDef(x.defId); return def.effects.some((e: any) => e.kind === 'gainAp' && e.amount >= def.cost) }); if (free) return free
  if (canKillNow(state, affordable)) { const d = byDmg(affordable)[0]; if (d) return d }
  if ((lethal || dangerous) && !guarded) { const b = bestBlock(); if (b) return b; const db = bestDebuff(); if (db) return db; const h = bestHeal(); if (h) return h }
  if (lethal) { const h = bestHeal(); if (h) return h }
  const safe = affordable.filter((x) => selfDmgOf(x) === 0 || state.player.hp - selfDmgOf(x) - Math.max(0, inc - state.player.block) > 0)
  const d = byDmg(safe)[0]; if (d) return d
  const u = bestUtil(); if (u) return u
  const h = bestHeal(); if (h && state.player.hp < state.player.maxHp * 0.7) return h
  const b = bestBlock(); if (b && !guarded && inc > 0) return b
  return null
}
function pickPlanner(state: GameState, affordable: CardInstance[]): CardInstance | null {
  const free = affordable.find((x) => { const def = getCardDef(x.defId); return def.effects.some((e: any) => e.kind === 'gainAp' && e.amount >= def.cost) }); if (free) return free
  if (canKillNow(state, affordable)) { const d = affordable.filter((x) => dmgOf(x) > 0).sort((p, q) => totalDmg(state, q) / costOf(q) - totalDmg(state, p) / costOf(p))[0]; if (d) return d }
  const ctx = plannerCtx(state); const scored = affordable.map((x) => ({ x, ...plannerValue(state, x, ctx), c: costOf(x) }))
  if (ctx.hpAfterNow <= 0) { const saving = scored.filter((s) => s.def > 0).sort((p, q) => q.def - p.def || q.v / q.c - p.v / p.c); if (saving.length) return saving[0].x }
  scored.sort((p, q) => q.v / q.c - p.v / p.c); return scored[0].v <= 0 ? null : scored[0].x
}
const guideAp = () => DIVINATION_CHOICES[1].effects.filter((e): e is Extract<Effect, { kind: 'gainAp' }> => e.kind === 'gainAp').reduce((s, e) => s + e.amount, 0)
const guideDraw = () => DIVINATION_CHOICES[1].effects.filter((e): e is Extract<Effect, { kind: 'draw' }> => e.kind === 'draw').reduce((s, e) => s + e.amount, 0)
function oracleEv(state: GameState) {
  const inc = effIncoming(state), rawInc = incomingOf(state)
  const guardGain = Math.max(RULES.divination.guardMin, Math.floor(rawInc * RULES.divination.guardRatio))
  const hpAfter = projectedHp(state), unblocked = Math.max(0, inc - state.player.block), saved = Math.min(guardGain, unblocked), lowHp = state.player.hp <= state.player.maxHp * 0.35
  const guard = hpAfter <= 0 && saved > 0 && hpAfter + saved > 0 ? 1000 : saved * (lowHp ? 2 : 1)
  const strike = state.enemy.hp + state.enemy.block <= 4 ? 1000 : 4 + (state.round >= 5 ? 1 : 0)
  const ap = state.ap.current + guideAp(); const newly = state.hand.filter((x) => costOf(x) > state.ap.current && costOf(x) <= ap)
  let best = 0; for (const x of newly) { const v = cardValue(state, x); if (v > best) best = v }
  const drawEv = guideDraw() * (ap - (newly.length ? Math.min(...newly.map(costOf)) : 0) >= 1 ? 2.5 : 1.0)
  return { guard, guide: best + drawEv, strike }
}
function pickOracle(state: GameState, policy: Policy): number {
  if (state.divination.remaining <= 0 || state.divination.usedThisRound) return -1
  if (policy === 'naive' || policy === 'greedy') return 2
  const inc = effIncoming(state), hpAfter = projectedHp(state), lethal = hpAfter <= 0, dangerous = hpAfter < state.player.maxHp * 0.35, roundsLeft = RULES.totalRounds - state.round + 1
  if (state.enemy.hp + state.enemy.block <= 4) return 2
  if (policy === 'planner') {
    const ev = oracleEv(state); const ult = upcomingUltimates(state)
    const mustHold = state.divination.remaining <= ult && !lethal && !dangerous && roundsLeft > state.divination.remaining; const dump = roundsLeft <= state.divination.remaining
    const best = Math.max(ev.guard, ev.guide, ev.strike); const choice = best === ev.guard ? 0 : best === ev.guide ? 1 : 2
    if (best >= 1000) return choice; if (mustHold) return -1; if (dump) return choice; if (best >= 8) return choice
    if ((lethal || dangerous) && inc > 0 && ev.guard > 0) return 0
    return -1
  }
  if (lethal && inc > 0) return 0
  if (state.divination.remaining >= roundsLeft) return dangerous && inc > 0 ? 0 : 2
  if (dangerous && inc > 0) return 0
  if (roundsLeft <= state.divination.remaining) return inc >= 10 && state.player.block < inc ? 0 : 2
  return -1
}
type Stake = { level: number; choice?: 'race' | 'pressure' | 'tempo' }
type Rec = { status: string; round: number; hp: number; minHp: number; bursts: number; roundsPlayed: number; playedPerRound: number[]; unusedApPerRound: number[]; handPerRound: number[]; ultR: number; ultGuarded: number; score: number; oracleUses: number; guardUses: number }
function play(godId: GodId, enemyId: EnemyId, diff: Difficulty, policy: Policy, seed: number, stake: Stake, daily = false): Rec {
  const deck = getRecommendedDeck(godId)
  let { state } = applyAction(null, { type: 'START_GAME', seed: `p254-${seed}`, godId, enemyId, deck, difficulty: diff, ...(daily ? { mode: 'daily' as const, dailyKey: '2026-10-03', modifier: RULES.daily.modifier } : {}), ...(stake.level ? { stake: stake.level, ...(stake.choice ? { stakeChoice: stake.choice } : {}) } : {}) } as any)
  const rnd = mulberry32(seed * 7919 + godId.length * 131 + enemyId.length * 17 + 1)
  let minHp = state.player.hp, bursts = 0, ultR = 0, ultGuarded = 0, score = 0, oracleUses = 0, guardUses = 0
  const playedPerRound: number[] = [], unusedApPerRound: number[] = [], handPerRound: number[] = []
  let guard = 0
  while (state.status === 'playing' && guard < 600) {
    guard++
    const r = state.round; const ult = isUlt(state.enemy.intent); handPerRound.push(state.hand.length)
    let played = 0, inner = 0
    while (state.status === 'playing' && inner < 60) {
      inner++
      const affordable = state.hand.filter((x) => costOf(x) <= state.ap.current)
      if (affordable.length) {
        const card = policy === 'randomRO' ? affordable[Math.floor(rnd() * affordable.length)] : policy === 'planner' ? pickPlanner(state, affordable) : pickReader(state, policy, affordable)
        if (card) { const res = applyAction(state, { type: 'PLAY_CARD', uid: card.uid }); for (const e of res.events) { if (e.t === 'RESONANCE_BURST') bursts++; else if (e.t === 'GAME_ENDED') score = e.totalScore }; state = res.state; played++; continue }
      }
      const o = pickOracle(state, policy === 'randomRO' ? 'reader' : policy)
      if (o >= 0) { const res = applyAction(state, { type: 'USE_DIVINATION', choiceIndex: o }); for (const e of res.events) { if (e.t === 'RESONANCE_BURST') bursts++; else if (e.t === 'GAME_ENDED') score = e.totalScore }; state = res.state; oracleUses++; if (o === 0) guardUses++; continue }
      break
    }
    playedPerRound.push(played)
    if (state.status !== 'playing') break
    if (ult) { ultR++; if (state.player.block >= effIncoming(state) * 0.5) ultGuarded++ }
    unusedApPerRound.push(state.ap.current)
    const res = applyAction(state, { type: 'END_ROUND' }); for (const e of res.events) { if (e.t === 'RESONANCE_BURST') bursts++; else if (e.t === 'GAME_ENDED') score = e.totalScore }; state = res.state
    minHp = Math.min(minHp, state.player.hp)
    void r
  }
  return { status: state.status, round: state.round, hp: state.player.hp, minHp, bursts, roundsPlayed: unusedApPerRound.length, playedPerRound, unusedApPerRound, handPerRound, ultR, ultGuarded, score, oracleUses, guardUses }
}
const mean = (xs: number[]) => (xs.length ? xs.reduce((p, q) => p + q, 0) / xs.length : 0)
function aggregate(recs: Rec[]) {
  const n = recs.length; const won = recs.filter((r) => r.status === 'won')
  const perR = (sel: (r: Rec) => number[]) => [0, 1, 2, 3, 4, 5, 6].map((i) => mean(recs.filter((r) => sel(r).length > i).map((r) => sel(r)[i])))
  return { n, win: won.length / n, lost: mean(recs.map((r) => +(r.status === 'lost'))), unfinished: mean(recs.map((r) => +(r.status === 'finished'))), killR: mean(won.map((r) => r.round)), reachR6: mean(recs.map((r) => +(r.roundsPlayed >= 5 && (r.status !== 'won' || r.round >= 6)))), minHp: mean(recs.map((r) => r.minHp)), hpWon: mean(won.map((r) => r.hp)), gs: mean(recs.map((r) => +(r.bursts > 0))), scoreWon: mean(won.map((r) => r.score)), oracleUses: mean(recs.map((r) => r.oracleUses)), guardShare: mean(recs.map((r) => (r.oracleUses ? r.guardUses / r.oracleUses : 0))), ultReach: mean(recs.map((r) => +(r.ultR > 0))), ultGuardRate: mean(recs.filter((r) => r.ultR > 0).map((r) => r.ultGuarded / r.ultR)), playedPerRound: perR((r) => r.playedPerRound), unusedApPerRound: perR((r) => r.unusedApPerRound), handPerRound: perR((r) => r.handPerRound) }
}
const LEVELS: { key: string; stake: Stake }[] = [{ key: '1', stake: { level: 1 } }, { key: '3', stake: { level: 3 } }, { key: '5', stake: { level: 5 } }, { key: '6', stake: { level: 6 } }, { key: '7p', stake: { level: 7, choice: 'pressure' } }, { key: '7r', stake: { level: 7, choice: 'race' } }, { key: '7t', stake: { level: 7, choice: 'tempo' } }]

          const recs: Rec[] = []; const byEnemy: Record<string, Rec[]> = {}; const byGod: Record<string, Rec[]> = {}; const cells: Record<string, number> = {}
          for (const god of GODS) for (const enemy of ENEMIES) { const cr: Rec[] = []; for (let i = 0; i < seeds; i++) { const r = play(god.id, enemy.id, diff, policy, i, stake, daily); recs.push(r); (byEnemy[enemy.name] ??= []).push(r); (byGod[god.nameJa] ??= []).push(r); cr.push(r); games++ } cells[`${god.nameJa}×${enemy.name}`] = mean(cr.map((r) => +(r.status === 'won'))) }
          out[vid][face][policy] = { agg: aggregate(recs), byEnemy: Object.fromEntries(Object.entries(byEnemy).map(([k, v]) => { const a = aggregate(v); return [k, { win: a.win, killR: a.killR, unfinished: a.unfinished, gs: a.gs }] })), byGod: Object.fromEntries(Object.entries(byGod).map(([k, v]) => { const a = aggregate(v); return [k, { win: a.win, killR: a.killR, unfinished: a.unfinished, gs: a.gs, scoreWon: a.scoreWon }] })), cells }

    }
    restoreAp()
    writeFileSync(`${OUT}/sim260${process.env.D260_TAG ? '-' + process.env.D260_TAG : ''}.json`, JSON.stringify({ runtime: '538a3ef', date: new Date().toISOString(), games, elapsed: (Date.now() - t0) / 1000, seeds: SEEDS, seedsRank: SEEDS_RANK, variants: VARIANTS, out }))

  }, 7_200_000)
})
