 * 決定259 — 原因分解の対照（D*）と是正候補（C*）を **プロセス内のデータ差し替え** で表現する共通モジュール。

 *   GODS[].resonanceEffects / GODS[].passive, GOD_PASSIVE_HOOKS（得意技の追加）

 *   NO_STAKE_RULES.divinationCount
 *   デッキ（変数差し替え：`deckOverride(godId)` が undefined なら推奨デッキ）

type Mutable<T> = { -readonly [K in keyof T]: T[K] }
const rulesM = RULES as unknown as {

// ---------------------------------------------------------------------------
// snapshot / restore
// ---------------------------------------------------------------------------
const deepCopy = <T,>(x: T): T => JSON.parse(JSON.stringify(x)) as T
const PROD = {
  resonanceEffects: new Map(GODS.map((g) => [g.id, deepCopy(g.resonanceEffects)])),
  passive: new Map(GODS.map((g) => [g.id, g.passive ? deepCopy(g.passive) : undefined])),
  otomoEffects: new Map(OTOMOS.map((o) => [o.id, deepCopy(o.effectsByForm)])),
  cardEffects: new Map(ALL_CARDS.map((c) => [c.id, deepCopy(c.effects)])),
  cardBonus: new Map(ALL_CARDS.map((c) => [c.id, c.bonus ? deepCopy(c.bonus) : undefined])),
  tempoByRound: [...RULES.score.tempoByRound],
  comboSteps: [...RULES.score.comboSteps],
  comboOverflow: RULES.score.comboOverflow,
  resonanceMax: RULES.resonance.max,
  passivesEnabled: RULES.godIdentity.passivesEnabled,
  cardBonusEnabled: RULES.godIdentity.cardBonusEnabled,
  fukueiHpRatio: RULES.godPassive.fukuei.hpRatio,
  divinationCount: NO_STAKE_RULES.divinationCount,
  hookKeys: Object.keys(GOD_PASSIVE_HOOKS),
}

let deckOverrideFn: ((godId: GodId) => CardDefId[] | undefined) | null = null
export function deckOverride(godId: GodId): CardDefId[] {
  return (deckOverrideFn && deckOverrideFn(godId)) ?? getRecommendedDeck(godId)
}

export function restore(): void {
  for (const g of GODS) {
    ;(g as { resonanceEffects: Effect[] }).resonanceEffects = deepCopy(PROD.resonanceEffects.get(g.id)!)
    const p = PROD.passive.get(g.id)
    if (p) (g as { passive?: unknown }).passive = deepCopy(p)
    else delete (g as { passive?: unknown }).passive
  }
  for (const o of OTOMOS) (o as { effectsByForm: Record<OtomoForm, Effect[]> }).effectsByForm = deepCopy(PROD.otomoEffects.get(o.id)!)
  for (const c of ALL_CARDS) {
    ;(c as { effects: Effect[] }).effects = deepCopy(PROD.cardEffects.get(c.id)!)
    const b = PROD.cardBonus.get(c.id)
    if (b) (c as { bonus?: unknown }).bonus = deepCopy(b)
    else delete (c as { bonus?: unknown }).bonus
  }
  rulesM.score.tempoByRound = [...PROD.tempoByRound]
  rulesM.score.comboSteps = [...PROD.comboSteps]
  rulesM.score.comboOverflow = PROD.comboOverflow
  rulesM.resonance.max = PROD.resonanceMax
  rulesM.godIdentity.passivesEnabled = PROD.passivesEnabled
  rulesM.godIdentity.cardBonusEnabled = PROD.cardBonusEnabled
  rulesM.godPassive.fukuei.hpRatio = PROD.fukueiHpRatio
  ;(NO_STAKE_RULES as { divinationCount: number }).divinationCount = PROD.divinationCount
  for (const k of Object.keys(hooksM)) if (!PROD.hookKeys.includes(k)) delete hooksM[k]
  deckOverrideFn = null
}

export function assertRestored(): void {
  for (const g of GODS) {
    expect(g.resonanceEffects).toEqual(PROD.resonanceEffects.get(g.id))
    expect(g.passive).toEqual(PROD.passive.get(g.id))
  }
  for (const o of OTOMOS) expect(o.effectsByForm).toEqual(PROD.otomoEffects.get(o.id))
  for (const c of ALL_CARDS) {
    expect(c.effects).toEqual(PROD.cardEffects.get(c.id))
    expect(c.bonus).toEqual(PROD.cardBonus.get(c.id))
  }
  expect([...RULES.score.tempoByRound]).toEqual(PROD.tempoByRound)
  expect([...RULES.score.comboSteps]).toEqual(PROD.comboSteps)
  expect(RULES.score.comboOverflow).toBe(PROD.comboOverflow)
  expect(RULES.resonance.max).toBe(PROD.resonanceMax)
  expect(RULES.godIdentity.passivesEnabled).toBe(PROD.passivesEnabled)
  expect(RULES.godIdentity.cardBonusEnabled).toBe(PROD.cardBonusEnabled)
  expect(RULES.godPassive.fukuei.hpRatio).toBe(PROD.fukueiHpRatio)
  expect(NO_STAKE_RULES.divinationCount).toBe(PROD.divinationCount)
  expect(Object.keys(GOD_PASSIVE_HOOKS).sort()).toEqual([...PROD.hookKeys].sort())
  expect(deckOverrideFn).toBeNull()
  expect(deckOverride(GOD_IDS.saika)).toEqual(getRecommendedDeck(GOD_IDS.saika))
}

const god = (id: GodId) => GODS.find((g) => g.id === id)!
const card = (id: CardDefId) => getCardDef(id)
const setBurst = (id: GodId, effects: Effect[]) => { (god(id) as { resonanceEffects: Effect[] }).resonanceEffects = effects }
const patchBurst = (id: GodId, f: (e: Effect) => Effect | null) => {
  const g = god(id) as { resonanceEffects: Effect[] }
  g.resonanceEffects = g.resonanceEffects.map(f).filter((e): e is Effect => e !== null)
}
const patchCard = (id: CardDefId, f: (e: Effect) => Effect) => { (card(id) as { effects: Effect[] }).effects = card(id).effects.map(f) }
const setBonusWhen = (id: CardDefId, when: NonNullable<CardDef['bonus']>['when']) => { const c = card(id) as { bonus?: CardDef['bonus'] }; if (c.bonus) c.bonus = { ...c.bonus, when } }
const isEnemyDebuff = (e: Effect): e is Extract<Effect, { kind: 'debuff' }> => e.kind === 'debuff' && e.target === 'enemy'

/**
 * 推奨デッキの専用カードを、同じ種別の共通カードへ置換したデッキ（D2）。
 * 置換先は「その種別の共通カードのうち、デッキ内の枚数が最少のもの（同数なら定義順）」。1 種 2 枚上限を守る。
 */
export function commonizedDeck(godId: GodId): CardDefId[] {
  const deck = [...getRecommendedDeck(godId)]
  const commons = getCardPoolForGod(godId).filter((c) => !c.godId)
  const count = new Map<CardDefId, number>()
  for (const id of deck) if (!card(id).godId) count.set(id, (count.get(id) ?? 0) + 1)
  return deck.map((id) => {
    const def = card(id)
    if (!def.godId) return id
    const cands = commons.filter((c) => c.type === (def.type as CardType) && (count.get(c.id) ?? 0) < RULES.deckBuilding.maxCopiesPerCard)
    const pick = cands.sort((a, b) => (count.get(a.id) ?? 0) - (count.get(b.id) ?? 0))[0] ?? commons.find((c) => (count.get(c.id) ?? 0) < RULES.deckBuilding.maxCopiesPerCard)!
    count.set(pick.id, (count.get(pick.id) ?? 0) + 1)
    return pick.id
  })
}

/** デッキの 1 枚入れ替え（from の最初の 1 枚を to に）。1 種 2 枚上限は呼び出し側で担保 */
export function swapDeck(deck: CardDefId[], swaps: [CardDefId, CardDefId][]): CardDefId[] {
  const d = [...deck]
  for (const [from, to] of swaps) { const i = d.indexOf(from); if (i < 0) throw new Error(); d[i] = to }
  return d
}

/** 複合候補（単独結果を見てから追加する。id は "X+Y" の形で登録） */
export function registerCombo(ids: string[], label?: string): Variant {
  const id = ids.join('+')
  const v: Variant = { id, label: label ?? ids.map((x) => VARIANTS[x].label).join(' ＋ '), group: 'combo', apply: () => { for (const x of ids) VARIANTS[x].apply() } }
  VARIANTS[id] = v
  return v
}

export function applyVariant(id: string): Variant {
  restore()
  const v = VARIANTS[id]
  if (!v) throw new Error(`unknown variant ${id}`)
  v.apply()
  return v
}

/** 環境変数 D259_VARIANTS（カンマ区切り）。"A+B" は複合として自動登録 */
export function variantIdsFromEnv(defaults: string[]): string[] {
  const raw = process.env.D259_VARIANTS
  const ids = raw ? raw.split(',').map((s) => s.trim()).filter(Boolean) : defaults
  for (const id of ids) if (!VARIANTS[id] && id.includes('+')) registerCombo(id.split('+'))
  return ids
}
