// ---------------- policies (same as Decision244 harness, reader oracle rule generalized) ----------------
type Policy = 'reader' | 'readerBurn' | 'naive' | 'greedy'
const eff = (def: CardDef, kind: string, target?: string) =>
  def.effects.filter((e: any) => e.kind === kind && (target ? e.target === target : true)).reduce((s, e: any) => s + (e.amount ?? 0), 0)
const dmgOf = (x: CardInstance) => eff(getCardDef(x.defId), 'damage', 'enemy')
const blockOf = (x: CardInstance) => eff(getCardDef(x.defId), 'block')
const healOf = (x: CardInstance) => eff(getCardDef(x.defId), 'heal')
const selfDmgOf = (x: CardInstance) => eff(getCardDef(x.defId), 'damage', 'self')
const debuffOf = (x: CardInstance) => getCardDef(x.defId).effects.filter((e: any) => e.kind === 'debuff' && e.target === 'enemy').reduce((s, e: any) => s + e.amount, 0)
const costOf = (x: CardInstance) => getCardDef(x.defId).cost + (x.costModifier ?? 0)
const utilOf = (x: CardInstance) => getCardDef(x.defId).effects.reduce((s, e: any) => {
  if (e.kind === 'draw') return s + e.amount * 4
  if (e.kind === 'gainAp') return s + e.amount * 4
  if (e.kind === 'resonance') return s + e.amount * 2.5
  if (e.kind === 'buff' && e.target === 'self') return s + e.amount * e.rounds
  if (e.kind === 'debuff' && e.target === 'enemy') return s + e.amount * e.rounds
  return s
}, 0)
const incomingOf = (s: GameState) => (s.enemy.intent ? enemyActionTotal(s.enemy.intent) : 0)
const effIncoming = (s: GameState) => Math.max(0, incomingOf(s) + sumBuff(s.enemy.buffs, 'atk'))
const projectedHp = (s: GameState) => s.player.hp - Math.max(0, effIncoming(s) - s.player.block)
function bonusDmgIf(state: GameState, x: CardInstance): number {
  const def = getCardDef(x.defId)
  if (!def.bonus) return 0
  const extra = def.bonus.effects.filter((e: any) => e.kind === 'damage' && e.target === 'enemy').reduce((s, e: any) => s + e.amount, 0)
  if (extra === 0) return 0
  const inc = incomingOf(state)
  switch (def.bonus.when) {
    case 'combo': return state.cardsPlayedThisRound >= 1 ? extra : 0
    case 'charged': return state.resonance.value >= RULES.cardBonus.chargedThreshold ? extra : 0
    case 'enemyBig': return inc >= RULES.cardBonus.enemyBigThreshold ? extra : 0
    case 'lowHp': return state.player.hp <= Math.floor(state.player.maxHp * 0.5) ? extra : 0
    case 'blocked': return inc > 0 && state.player.block + blockOf(x) >= inc ? extra : 0
  }
  return 0
}
const totalDmg = (s: GameState, x: CardInstance) => dmgOf(x) + bonusDmgIf(s, x) + (dmgOf(x) > 0 ? Math.max(0, sumBuff(s.player.buffs, 'atk')) : 0)
function canKillNow(state: GameState, affordable: CardInstance[]): boolean {
  const dmgCards = affordable.filter((x) => dmgOf(x) > 0).sort((p, q) => totalDmg(state, q) / costOf(q) - totalDmg(state, p) / costOf(p))
  let ap = state.ap.current
  let sum = 0
  for (const x of dmgCards) if (costOf(x) <= ap) { ap -= costOf(x); sum += totalDmg(state, x) }
  return sum >= state.enemy.hp + state.enemy.block
}
let intentDecisions = 0 // counts reader picks taken because of lethal/dangerous intent
function pickCard(state: GameState, policy: Policy): CardInstance | null {
  const affordable = state.hand.filter((x) => costOf(x) <= state.ap.current)
  if (affordable.length === 0) return null
  if (policy === 'naive') return affordable[0]
  const inc = effIncoming(state)
  const hpAfter = projectedHp(state)
  const lethal = hpAfter <= 0
  const dangerous = hpAfter < state.player.maxHp * 0.35
  const guarded = state.player.block >= inc
  const byDmg = (arr: CardInstance[]) => arr.filter((x) => dmgOf(x) > 0).sort((p, q) => totalDmg(state, q) / costOf(q) - totalDmg(state, p) / costOf(p))
  const bestBlock = () => affordable.filter((x) => blockOf(x) > 0).sort((p, q) => blockOf(q) - blockOf(p))[0]
  const bestDebuff = () => affordable.filter((x) => debuffOf(x) > 0).sort((p, q) => debuffOf(q) - debuffOf(p))[0]
  const bestHeal = () => affordable.filter((x) => healOf(x) > 0).sort((p, q) => healOf(q) - healOf(p))[0]
  const bestUtil = () => affordable.filter((x) => utilOf(x) > 0 && dmgOf(x) === 0 && blockOf(x) === 0 && healOf(x) === 0).sort((p, q) => utilOf(q) / costOf(q) - utilOf(p) / costOf(p))[0]
  if (policy === 'greedy') {
    const d = byDmg(affordable)[0]
    if (d) return d
    return bestUtil() ?? bestBlock() ?? bestHeal() ?? affordable.sort((p, q) => costOf(q) - costOf(p))[0]
  }
  const free = affordable.find((x) => { const def = getCardDef(x.defId); return def.effects.some((e: any) => e.kind === 'gainAp' && e.amount >= def.cost) })
  if (free) return free
  if (canKillNow(state, affordable)) { const d = byDmg(affordable)[0]; if (d) return d }
  if ((lethal || dangerous) && !guarded) {
    const b = bestBlock(); if (b) { intentDecisions++; return b }
    const db = bestDebuff(); if (db) { intentDecisions++; return db }
    const h = bestHeal(); if (h) { intentDecisions++; return h }
  }
  if (lethal) { const h = bestHeal(); if (h) { intentDecisions++; return h } }
  const safe = affordable.filter((x) => selfDmgOf(x) === 0 || state.player.hp - selfDmgOf(x) - Math.max(0, inc - state.player.block) > 0)
  const d = byDmg(safe)[0]
  if (d) return d
  const u = bestUtil(); if (u) return u
  const h = bestHeal(); if (h && state.player.hp < state.player.maxHp * 0.7) return h
  const b = bestBlock(); if (b && !guarded && inc > 0) return b
  return null
}
/** -1 = skip. Engine cap is state.divination.remaining; reader saves when remaining < rounds left. */
function pickOracle(state: GameState, policy: Policy): number {
  if (state.divination.remaining <= 0 || state.divination.usedThisRound) return -1
  if (policy === 'naive' || policy === 'greedy') return 2
  const inc = effIncoming(state)
  const hpAfter = projectedHp(state)
  const lethal = hpAfter <= 0
  const dangerous = hpAfter < state.player.maxHp * 0.35
  const kill = state.enemy.hp + state.enemy.block <= 4
  if (kill) return 2
  if (lethal && inc > 0) return 0
  const roundsLeft = RULES.totalRounds - state.round + 1
  if (policy === 'readerBurn' || state.divination.remaining >= roundsLeft) {
    if (dangerous && inc > 0) return 0
    return 2
  }
  if (dangerous && inc > 0) return 0
  if (roundsLeft <= state.divination.remaining) return inc >= 10 && state.player.block < inc ? 0 : 2
  return -1
}

// ---------------- game loop ----------------
type Game = {
  god: string; enemy: string; diff: Difficulty; policy: Policy; ov: OracleVariant; ev: EnemyVariant; seed: number
  status: string; round: number; hp: number; minHp: number
  roundsAlive: number; reachR5: boolean; reachR6: boolean; reachR7: boolean
  oracleR: number[]; oracleT: number[]; burstR: number[]
  bigIntentRounds: number; peakEncountered: boolean; peakDefended: boolean
  intentDecisions: number
  perRound: { r: number; inActual: number; hpEnd: number; status: string }[]
}
function playOne(godId: GodId, enemyId: EnemyId, diff: Difficulty, policy: Policy, ov: OracleVariant, ev: EnemyVariant, seed: number, label: { god: string; enemy: string; peakRound: number }): Game {
  const deck = getRecommendedDeck(godId)
  let { state } = applyAction(null, { type: 'START_GAME', seed: `d245-${seed}`, godId, enemyId, deck, difficulty: diff })
  const oracleR: number[] = []
  const oracleT: number[] = []
  const burstR: number[] = []
  const perRound: Game['perRound'] = []
  let minHp = state.player.hp
  let bigIntentRounds = 0
  let peakEncountered = false
  let peakDefended = false
  intentDecisions = 0
  let guard = 0
  while (state.status === 'playing' && guard < 600) {
    guard++
    const r = state.round
    if (incomingOf(state) >= 10) bigIntentRounds++
    if (r === label.peakRound) peakEncountered = true
    let inner = 0
    while (state.status === 'playing' && inner < 60) {
      inner++
      const card = pickCard(state, policy)
      if (card) {
        const res = applyAction(state, { type: 'PLAY_CARD', uid: card.uid })
        if (res.events.some((e) => e.t === 'RESONANCE_BURST')) burstR.push(r)
        state = res.state
        continue
      }
      const o = pickOracle(state, policy)
      if (o >= 0) {
        const res = applyAction(state, { type: 'USE_DIVINATION', choiceIndex: o })
        if (res.events.some((e) => e.t === 'RESONANCE_BURST')) burstR.push(r)
        state = res.state
        oracleR.push(r)
        oracleT.push(o)
        continue
      }
      break
    }
    const hpBefore = state.player.hp
    if (r === label.peakRound && state.status === 'playing') peakDefended = state.player.block >= effIncoming(state) * 0.5
    if (state.status === 'playing') {
      const res = applyAction(state, { type: 'END_ROUND' })
      if (res.events.some((e) => e.t === 'RESONANCE_BURST')) burstR.push(r)
      state = res.state
    }
    minHp = Math.min(minHp, state.player.hp)
    perRound.push({ r, inActual: Math.max(0, hpBefore - state.player.hp), hpEnd: state.player.hp, status: state.status })
    if (state.status !== 'playing') break
  }
  const roundsAlive = perRound.length
  return {
    god: label.god, enemy: label.enemy, diff, policy, ov, ev, seed,
    status: state.status, round: state.round, hp: state.player.hp, minHp,
    roundsAlive, reachR5: roundsAlive >= 5, reachR6: roundsAlive >= 6, reachR7: roundsAlive >= 7,
    oracleR, oracleT, burstR, bigIntentRounds, peakEncountered, peakDefended, intentDecisions, perRound,
  }
}

    // 5. clear round distribution (reader normal/hard) per variant
    md.push('', '## 5. Clear-round distribution (reader)', '', '| enemy | oracle | diff | R1 | R2 | R3 | R4 | R5 | R6 | R7 | lost | unfinished |', '|---|---|---|---|---|---|---|---|---|---|---|---|')
    for (const diff of ['normal', 'hard'] as Difficulty[]) for (const ev of EVS) for (const ov of OVS) {
      const r = by((x) => x.ev === ev && x.ov === ov && x.diff === diff && x.policy === 'reader')
      const cnt = (k: number) => r.filter((x) => x.status === 'won' && x.round === k).length
      md.push(`| ${ev} | ${ov} | ${diff} | ${[1, 2, 3, 4, 5, 6, 7].map((k) => cnt(k)).join(' | ')} | ${r.filter((x) => x.status === 'lost').length} | ${r.filter((x) => x.status === 'finished').length} |`)
    }

    // 6. per-enemy & per-god tables for candidate cells (normal & hard; reader/naive/greedy)
    md.push('', '## 6. Per enemy (reader win / naive win / greedy win / reader dmg-min HP ratio)', '')
    for (const diff of ['normal', 'hard'] as Difficulty[]) for (const ev of EVS) for (const ov of ['A7', 'B3', 'C2'] as OracleVariant[]) {
      md.push(`### ${diff} ${ev} ${ov}`, '', '| enemy | reader win | reader lost | reader unfinished | naive win | greedy win | reader winR | peak encountered | min HP ratio |', '|---|---|---|---|---|---|---|---|---|')
      for (const enemy of ENEMIES) {
        const g = (p: Policy) => by((x) => x.ev === ev && x.ov === ov && x.diff === diff && x.policy === p && x.enemy === enemy.name)
        const r = g('reader')
        md.push(`| ${enemy.name} | ${pct(win(r))} | ${pct(lost(r))} | ${pct(fin(r))} | ${pct(win(g('naive')))} | ${pct(win(g('greedy')))} | ${f2(winR(r))} | ${pct(mean(r.map((x) => +x.peakEncountered)))} | ${f2(mean(r.map((x) => x.minHp / (diff === 'hard' ? 27 : 30))))} |`)
      }
      md.push('')
    }
    md.push('', '## 7. Per god (reader win / naive win / greedy win)', '')
    for (const diff of ['normal', 'hard'] as Difficulty[]) for (const ev of EVS) for (const ov of ['A7', 'B3', 'C2'] as OracleVariant[]) {
      md.push(`### ${diff} ${ev} ${ov}`, '', '| god | reader win | reader lost | reader unfinished | naive win | greedy win | reader winR |', '|---|---|---|---|---|---|---|')
      for (const god of GODS) {
        const g = (p: Policy) => by((x) => x.ev === ev && x.ov === ov && x.diff === diff && x.policy === p && x.god === god.nameJa)
        const r = g('reader')
        md.push(`| ${god.nameJa} | ${pct(win(r))} | ${pct(lost(r))} | ${pct(fin(r))} | ${pct(win(g('naive')))} | ${pct(win(g('greedy')))} | ${f2(winR(r))} |`)
      }
      md.push('')
    }
    // 8. god×enemy outliers for each candidate (normal/hard): cells with reader win <60% or unfinished >20% or naive win >=95%
    md.push('', '## 8. God × Enemy outliers (reader win < 60% or unfinished > 20%, or naive win ≥ 95% on normal)', '')
    for (const diff of ['normal', 'hard'] as Difficulty[]) for (const ev of EVS) for (const ov of ['A7', 'B3', 'C2'] as OracleVariant[]) {
      const rows: string[] = []
      for (const god of GODS) for (const enemy of ENEMIES) {
        const r = by((x) => x.ev === ev && x.ov === ov && x.diff === diff && x.policy === 'reader' && x.god === god.nameJa && x.enemy === enemy.name)
        const n = by((x) => x.ev === ev && x.ov === ov && x.diff === diff && x.policy === 'naive' && x.god === god.nameJa && x.enemy === enemy.name)
        const flag: string[] = []
        if (win(r) < 0.6) flag.push(`reader ${pct(win(r))}`)
        if (fin(r) > 0.2) flag.push(`unfinished ${pct(fin(r))}`)
        if (diff === 'normal' && win(n) >= 0.95) flag.push(`naive ${pct(win(n))}`)
        if (flag.length) rows.push(`- ${god.nameJa} × ${enemy.name}: ${flag.join(' / ')}`)
      }
      md.push(`### ${diff} ${ev} ${ov} — ${rows.length} cells`, ...(rows.length ? rows : ['- none']), '')
    }
    writeFileSync(`${OUT}/gate.md`, md.join('\n'))
    writeFileSync(`${OUT}/games-digest.txt`, require('node:crypto').createHash('sha256').update(JSON.stringify(games)).digest('hex'))
    console.log(`games=${games.length} elapsed=${elapsed}s`)
  }, 3_600_000)
})
