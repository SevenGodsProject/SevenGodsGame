// ---------------- helpers ----------------
type Policy = 'reader' | 'naive' | 'greedy' | 'oracleAware' | 'planner' | 'r_noGuard' | 'r_noWeaken' | 'r_noMend' | 'p_noGuard' | 'p_noWeaken' | 'p_noMend' | 'p_noTempo' | 'p_noAttune' | 'p_noEmpower'
const BAN: Partial<Record<Policy, CardSemantic>> = { r_noGuard: 'GUARD', r_noWeaken: 'WEAKEN', r_noMend: 'MEND', p_noGuard: 'GUARD', p_noWeaken: 'WEAKEN', p_noMend: 'MEND', p_noTempo: 'TEMPO', p_noAttune: 'ATTUNE', p_noEmpower: 'EMPOWER' }
const baseOf = (p: Policy): 'reader' | 'naive' | 'greedy' | 'oracleAware' | 'planner' => p.startsWith('r_') ? 'reader' : p.startsWith('p_') ? 'planner' : (p as any)
const isUlt = (x: EnemyActionDef | null | undefined) => !!x && (x.kind === 'special' || (x.kind === 'multiAttack' && !!x.special))
const eff = (def: CardDef, kind: string, target?: string) => def.effects.filter((e: any) => e.kind === kind && (target ? e.target === target : true)).reduce((s, e: any) => s + (e.amount ?? 0), 0)
const semOf = (x: CardInstance): CardSemantic => primarySemantic(getCardDef(x.defId).effects) ?? 'STRIKE'
const dmgOf = (x: CardInstance) => eff(getCardDef(x.defId), 'damage', 'enemy')
const blockOf = (x: CardInstance) => eff(getCardDef(x.defId), 'block')
const healOf = (x: CardInstance) => eff(getCardDef(x.defId), 'heal')
const selfDmgOf = (x: CardInstance) => eff(getCardDef(x.defId), 'damage', 'self')
const debuffOf = (x: CardInstance) => getCardDef(x.defId).effects.filter(isEnemyDebuff).reduce((s, e) => s + e.amount, 0)
const debuffRoundsOf = (x: CardInstance) => getCardDef(x.defId).effects.filter(isEnemyDebuff).reduce((s, e) => Math.max(s, e.rounds), 0)
const costOf = (x: CardInstance) => getCardDef(x.defId).cost + (x.costModifier ?? 0)
const utilOf = (x: CardInstance) => getCardDef(x.defId).effects.reduce((s, e: any) => {
  if (e.kind === 'draw' || e.kind === 'gainAp') return s + e.amount * 4
  if (e.kind === 'resonance') return s + e.amount * 2.5
  if ((e.kind === 'buff' && e.target === 'self') || (e.kind === 'debuff' && e.target === 'enemy')) return s + e.amount * e.rounds
  return s
}, 0)
const incomingOf = (s: GameState) => (s.enemy.intent ? enemyActionTotal(s.enemy.intent) : 0)
const effIncoming = (s: GameState) => Math.max(0, incomingOf(s) + sumBuff(s.enemy.buffs, 'atk'))
const projectedHp = (s: GameState) => s.player.hp - Math.max(0, effIncoming(s) - s.player.block)
function bonusDmgIf(state: GameState, x: CardInstance): number {
  const def = getCardDef(x.defId)
  if (!def.bonus) return 0
  const extra = def.bonus.effects.filter((e: any) => e.kind === 'damage' && e.target === 'enemy').reduce((s, e: any) => s + e.amount, 0)
  if (extra === 0) return 0
  const inc = incomingOf(state)
  switch (def.bonus.when) {
    case 'combo': return state.cardsPlayedThisRound >= 1 ? extra : 0
    case 'charged': return state.resonance.value >= RULES.cardBonus.chargedThreshold ? extra : 0
    case 'enemyBig': return inc >= RULES.cardBonus.enemyBigThreshold ? extra : 0
    case 'lowHp': return state.player.hp <= Math.floor(state.player.maxHp * 0.5) ? extra : 0
    case 'blocked': return inc > 0 && state.player.block + blockOf(x) >= inc ? extra : 0
  }
  return 0
}
const totalDmg = (s: GameState, x: CardInstance) => dmgOf(x) + bonusDmgIf(s, x) + (dmgOf(x) > 0 ? Math.max(0, sumBuff(s.player.buffs, 'atk')) : 0)
function maxDmgNow(state: GameState, affordable: CardInstance[]): number {
  const dmgCards = affordable.filter((x) => dmgOf(x) > 0).sort((p, q) => totalDmg(state, q) / costOf(q) - totalDmg(state, p) / costOf(p))
  let ap = state.ap.current, sum = 0
  for (const x of dmgCards) if (costOf(x) <= ap) { ap -= costOf(x); sum += totalDmg(state, x) }
  return sum
}
const canKillNow = (state: GameState, affordable: CardInstance[]) => maxDmgNow(state, affordable) >= state.enemy.hp + state.enemy.block
function upcomingUltimates(state: GameState): number {
  const def = ENEMIES.find((e) => e.id === state.enemy.defId)!
  let n = 0
  for (let r = state.round + 1; r <= RULES.totalRounds; r++) if (isUlt(def.actions[r - 1])) n++
  return n
}
/** future effective incoming for round r (raw table × current multiplier ratio, minus debuffs that will still be active) */
function futureIncoming(state: GameState, r: number): number {
  const def = ENEMIES.find((e) => e.id === state.enemy.defId)!
  const raw = enemyActionTotal(def.actions[r - 1])
  const rawNow = enemyActionTotal(def.actions[state.round - 1])
  const ratio = rawNow > 0 && state.enemy.intent ? incomingOf(state) / rawNow : 1
  const k = r - state.round
  const debuff = state.enemy.buffs.filter((b: any) => b.stat === 'atk' && b.amount < 0 && ((b.remainingRounds ?? b.rounds ?? 1) > k)).reduce((s: number, b: any) => s + b.amount, 0)
  return Math.max(0, Math.round(raw * ratio) + debuff)
}
function cardValue(state: GameState, x: CardInstance): number {
  const inc = effIncoming(state)
  const d = totalDmg(state, x)
  const b = Math.min(blockOf(x), Math.max(0, inc - state.player.block))
  const h = Math.min(healOf(x), state.player.maxHp - state.player.hp)
  const u = utilOf(x) * 0.5
  return d + b + h * 0.8 + u
}

// ---------------- reader (Lane 1 / D252 unchanged) ----------------
function pickCardReader(state: GameState, policy: Policy): CardInstance | null {
  let affordable = state.hand.filter((x) => costOf(x) <= state.ap.current)
  const ban = BAN[policy]; if (ban) affordable = affordable.filter((x) => semOf(x) !== ban)
  if (affordable.length === 0) return null
  const base = baseOf(policy)
  if (base === 'naive') return affordable[0]
  const inc = effIncoming(state)
  const hpAfter = projectedHp(state)
  const lethal = hpAfter <= 0
  const dangerous = hpAfter < state.player.maxHp * 0.35
  const guarded = state.player.block >= inc
  const byDmg = (arr: CardInstance[]) => arr.filter((x) => dmgOf(x) > 0).sort((p, q) => totalDmg(state, q) / costOf(q) - totalDmg(state, p) / costOf(p))
  const bestBlock = () => affordable.filter((x) => blockOf(x) > 0).sort((p, q) => blockOf(q) - blockOf(p))[0]
  const bestDebuff = () => affordable.filter((x) => debuffOf(x) > 0).sort((p, q) => debuffOf(q) * debuffRoundsOf(q) - debuffOf(p) * debuffRoundsOf(p))[0]
  const bestHeal = () => affordable.filter((x) => healOf(x) > 0).sort((p, q) => healOf(q) - healOf(p))[0]
  const bestUtil = () => affordable.filter((x) => utilOf(x) > 0 && dmgOf(x) === 0 && blockOf(x) === 0 && healOf(x) === 0).sort((p, q) => utilOf(q) / costOf(q) - utilOf(p) / costOf(p))[0]
  if (base === 'greedy') { const d = byDmg(affordable)[0]; if (d) return d; return bestUtil() ?? bestBlock() ?? bestHeal() ?? affordable.sort((p, q) => costOf(q) - costOf(p))[0] }
  const free = affordable.find((x) => { const def = getCardDef(x.defId); return def.effects.some((e: any) => e.kind === 'gainAp' && e.amount >= def.cost) })
  if (free) return free
  if (canKillNow(state, affordable)) { const d = byDmg(affordable)[0]; if (d) return d }
  if ((lethal || dangerous) && !guarded) { const b = bestBlock(); if (b) return b; const db = bestDebuff(); if (db) return db; const h = bestHeal(); if (h) return h }
  if (lethal) { const h = bestHeal(); if (h) return h }
  const safe = affordable.filter((x) => selfDmgOf(x) === 0 || state.player.hp - selfDmgOf(x) - Math.max(0, inc - state.player.block) > 0)
  const d = byDmg(safe)[0]; if (d) return d
  const u = bestUtil(); if (u) return u
  const h = bestHeal(); if (h && state.player.hp < state.player.maxHp * 0.7) return h
  const b = bestBlock(); if (b && !guarded && inc > 0) return b
  return null
}

// ---------------- planner (NEW) ----------------
/** HP-unit value of a card with a 3-round horizon; WEAKEN counts prevented damage over its duration, MEND counts recoverable HP,
 *  GUARD counts only this round's unblocked damage. Defensive weight rises with projected pressure. */
function plannerValue(state: GameState, x: CardInstance, ctx: { unblockedNow: number; pressure: number; hpAfterNow: number }): { v: number; def: number } {
  const dmg = totalDmg(state, x)
  const killable = state.enemy.hp + state.enemy.block
  let v = Math.min(dmg, killable) + (dmg >= killable && dmg > 0 ? 15 : 0)
  let def = 0
  def += Math.min(blockOf(x), ctx.unblockedNow)
  const db = debuffOf(x), dr = debuffRoundsOf(x)
  if (db > 0) {
    for (let k = 0; k < dr; k++) {
      const r = state.round + k
      if (r > RULES.totalRounds) break
      const target = k === 0 ? ctx.unblockedNow : futureIncoming(state, r)
      def += Math.min(db, Math.max(0, target)) * (k === 0 ? 1 : 0.9)
    }
  }
  const h = Math.min(healOf(x), state.player.maxHp - state.player.hp)
  def += h * (ctx.hpAfterNow < state.player.maxHp * 0.5 ? 1.0 : 0.6)
  const self = selfDmgOf(x)
  v += utilOf(x) * 0.5 - self * ctx.pressure
  return { v: v + def * ctx.pressure, def }
}
function pickCardPlanner(state: GameState, policy: Policy): CardInstance | null {
  let affordable = state.hand.filter((x) => costOf(x) <= state.ap.current)
  const ban = BAN[policy]; if (ban) affordable = affordable.filter((x) => semOf(x) !== ban)
  if (affordable.length === 0) return null
  const free = affordable.find((x) => { const def = getCardDef(x.defId); return def.effects.some((e: any) => e.kind === 'gainAp' && e.amount >= def.cost) })
  if (free) return free
  if (canKillNow(state, affordable)) { const d = affordable.filter((x) => dmgOf(x) > 0).sort((p, q) => totalDmg(state, q) / costOf(q) - totalDmg(state, p) / costOf(p))[0]; if (d) return d }
  const inc = effIncoming(state)
  const unblockedNow = Math.max(0, inc - state.player.block)
  const hpAfterNow = state.player.hp - unblockedNow
  // horizon pressure: HP after this round and the next two rounds (unguarded), relative to maxHp
  let future = 0
  for (let r = state.round + 1; r <= Math.min(RULES.totalRounds, state.round + 2); r++) future += futureIncoming(state, r)
  const hpHorizon = hpAfterNow - future
  const pressure = hpAfterNow <= 0 ? 3 : hpHorizon <= 0 ? 1.6 : hpHorizon < state.player.maxHp * 0.35 ? 1.2 : 0.6
  const scored = affordable.map((x) => ({ x, ...plannerValue(state, x, { unblockedNow, pressure, hpAfterNow }), c: costOf(x) }))
  if (hpAfterNow <= 0) {
    // must survive this round: prefer the card that saves the most now (def), tie-break by value per AP
    const saving = scored.filter((s) => s.def > 0).sort((p, q) => q.def - p.def || q.v / q.c - p.v / p.c)
    if (saving.length) return saving[0].x
  }
  scored.sort((p, q) => q.v / q.c - p.v / p.c)
  const best = scored[0]
  if (best.v <= 0) return null
  return best.x
}
function pickCard(state: GameState, policy: Policy): CardInstance | null {
  return baseOf(policy) === 'planner' ? pickCardPlanner(state, policy) : pickCardReader(state, policy)
}

// ---------------- oracle ----------------
const guideAp = () => DIVINATION_CHOICES[1].effects.filter((e): e is Extract<Effect, { kind: 'gainAp' }> => e.kind === 'gainAp').reduce((s, e) => s + e.amount, 0)
const guideDraw = () => DIVINATION_CHOICES[1].effects.filter((e): e is Extract<Effect, { kind: 'draw' }> => e.kind === 'draw').reduce((s, e) => s + e.amount, 0)
function oracleEv(state: GameState) {
  const inc = effIncoming(state), rawInc = incomingOf(state)
  const guardGain = Math.max(RULES.divination.guardMin, Math.floor(rawInc * RULES.divination.guardRatio))
  const hpAfter = projectedHp(state)
  const unblocked = Math.max(0, inc - state.player.block)
  const saved = Math.min(guardGain, unblocked)
  const lowHp = state.player.hp <= state.player.maxHp * 0.35
  const guard = hpAfter <= 0 && saved > 0 && hpAfter + saved > 0 ? 1000 : saved * (lowHp ? 2 : 1)
  const strike = state.enemy.hp + state.enemy.block <= 4 ? 1000 : 4 + (state.round >= 5 ? 1 : 0)
  const ap = state.ap.current + guideAp()
  const newly = state.hand.filter((x) => costOf(x) > state.ap.current && costOf(x) <= ap)
  let best = 0
  for (const x of newly) { const v = cardValue(state, x); if (v > best) best = v }
  const drawEv = guideDraw() * (ap - (newly.length ? Math.min(...newly.map(costOf)) : 0) >= 1 ? 2.5 : 1.0)
  return { guard, guide: best + drawEv, strike }
}
function pickOracle(state: GameState, policy: Policy): number {
  if (state.divination.remaining <= 0 || state.divination.usedThisRound) return -1
  const base = baseOf(policy)
  if (base === 'naive' || base === 'greedy') return 2
  const inc = effIncoming(state), hpAfter = projectedHp(state)
  const lethal = hpAfter <= 0, dangerous = hpAfter < state.player.maxHp * 0.35
  const roundsLeft = RULES.totalRounds - state.round + 1
  if (state.enemy.hp + state.enemy.block <= 4) return 2
  const guardOk = policy !== 'r_noGuard' && policy !== 'p_noGuard'
  if (base === 'oracleAware' || base === 'planner') {
    const ev = oracleEv(state)
    if (!guardOk) ev.guard = -1
    const ult = upcomingUltimates(state)
    const mustHold = state.divination.remaining <= ult && !lethal && !dangerous && roundsLeft > state.divination.remaining
    const dump = roundsLeft <= state.divination.remaining
    const best = Math.max(ev.guard, ev.guide, ev.strike)
    const choice = best === ev.guard ? 0 : best === ev.guide ? 1 : 2
    if (best >= 1000) return choice
    if (mustHold) return -1
    if (dump) return choice
    if (best >= 8) return choice
    if ((lethal || dangerous) && inc > 0 && ev.guard > 0) return 0
    return -1
  }
  if (lethal && inc > 0) return guardOk ? 0 : 2
  if (state.divination.remaining >= roundsLeft) return dangerous && inc > 0 && guardOk ? 0 : 2
  if (dangerous && inc > 0) return guardOk ? 0 : -1
  if (roundsLeft <= state.divination.remaining) return inc >= 10 && state.player.block < inc && guardOk ? 0 : 2
  return -1
}

        const sem = semOf(card), cost = costOf(card)
        const res = applyAction(state, { type: 'PLAY_CARD', uid: card.uid })
        for (const e of res.events) { if (e.t === 'RESONANCE_BURST') { bursts++; if (!burstR) burstR = r } else if (e.t === 'CARD_DRAWN') noteSeen(e.defId); else if (e.t === 'GAME_ENDED') score = e.totalScore }
        state = res.state
        played.push({ r, defId: card.defId, cost, sem })
        cnt[SEM_KEY[sem]] = (cnt[SEM_KEY[sem]] ?? 0) + 1
        apBySem[sem] = (apBySem[sem] ?? 0) + cost
        if (ultRound > 0 && r < ultRound) { preUltSemSet.add(SEM_KEY[sem]); if (sem === 'WEAKEN') preUltWeaken = true }
        continue
      }
      const o = pickOracle(state, policy)
      if (o >= 0) {
        const res = applyAction(state, { type: 'USE_DIVINATION', choiceIndex: o })
        for (const e of res.events) { if (e.t === 'RESONANCE_BURST') { bursts++; if (!burstR) burstR = r } else if (e.t === 'CARD_DRAWN') noteSeen(e.defId); else if (e.t === 'GAME_ENDED') score = e.totalScore }
        state = res.state
        uses.push({ r, t: o, ult })
        oracleSig = ['g', 'd', 's'][o]
        continue
      }
      break
    }
    const sig = SEMS.map((s) => (cnt[SEM_KEY[s]] ? `${SEM_KEY[s]}${cnt[SEM_KEY[s]]}` : '')).join('') + '|' + oracleSig
    roundSigs.push(sig)
    if (ult && !ultSig) {
      ultSig = sig
      // answer class at the ultimate round: how was the hit handled?
      const incNow = effIncoming(state)
      const debuffActive = sumBuff(state.enemy.buffs, 'atk') < 0
      const parts: string[] = []
      if (state.status !== 'playing') parts.push('KILL')
      if (oracleSig === 'g') parts.push('ORACLE')
      if (state.player.block > blockBefore) parts.push('GUARD')
      if (debuffActive) parts.push('WEAKEN')
      if ((cnt.M ?? 0) > 0) parts.push('MEND')
      if (parts.length === 0) parts.push(incNow - state.player.block <= 0 ? 'NONE' : 'TANK')
      ultAnswer = parts.join('+')
    }
    if (state.status !== 'playing') break
    if (ult) { ultR++; if (state.player.block >= effIncoming(state) * 0.5) ultGuarded++ }
    const hpBefore = state.player.hp
    const res = applyAction(state, { type: 'END_ROUND' })
    for (const e of res.events) { if (e.t === 'RESONANCE_BURST') { bursts++; if (!burstR) burstR = r } else if (e.t === 'CARD_DRAWN') noteSeen(e.defId); else if (e.t === 'GAME_ENDED') score = e.totalScore }
    state = res.state
    const taken = Math.max(0, hpBefore - state.player.hp)
    dmgTaken.push(taken)
    if (ult && ultDmg === null) ultDmg = taken
    minHp = Math.min(minHp, state.player.hp)
  }
  const status = state.status as Rec['status']
  let deathClass = 'won'
  if (status === 'lost') { const rr = dmgTaken.length; const act = enemyDef.actions[rr - 1]; deathClass = isUlt(act) ? 'ultimate' : rr < peakR ? 'pre-peak' : rr === peakR ? 'peak' : 'post-peak' }
  else if (status === 'finished') deathClass = burstR ? 'timeout-after-burst' : 'timeout-no-burst'
  return { status, round: state.round, hp: state.player.hp, minHp, uses, bursts, burstR, dmgTaken, deathClass, ultR, ultGuarded, played, seen, seenDef, planSig: roundSigs.join(' '), ultSig, ultAnswer, ultHandGuard, ultDmg, apBySem, roundsPlayed: dmgTaken.length, preUltWeaken, preUltSems: [...preUltSemSet].sort().join(''), score }
}

// ---------------- aggregation ----------------
const mean = (xs: number[]) => (xs.length ? xs.reduce((p, q) => p + q, 0) / xs.length : 0)
const std = (xs: number[]) => { if (xs.length < 2) return 0; const m = mean(xs); return Math.sqrt(mean(xs.map((x) => (x - m) ** 2))) }
const ratio = (num: number, den: number) => (den ? num / den : 0)
function gini(xs: number[]): number { const a = [...xs].sort((p, q) => p - q); const n = a.length; const s = a.reduce((p, q) => p + q, 0); if (!n || !s) return 0; let acc = 0; for (let i = 0; i < n; i++) acc += (2 * (i + 1) - n - 1) * a[i]; return acc / (n * s) }
function aggregate(recs: Rec[]) {
  const n = recs.length
  const won = recs.filter((r) => r.status === 'won'), lost = recs.filter((r) => r.status === 'lost'), fin = recs.filter((r) => r.status === 'finished')
  const uses = recs.flatMap((r) => r.uses)
  const byT = [0, 1, 2].map((t) => uses.filter((u) => u.t === t).length)
  const cls: Record<string, number> = {}
  for (const r of recs) if (r.status !== 'won') cls[r.deathClass] = (cls[r.deathClass] ?? 0) + 1
  const playsByDef: Record<string, number> = {}, seenByDef: Record<string, number> = {}
  let plays = 0, seen = 0
  for (const r of recs) { for (const p of r.played) { playsByDef[p.defId] = (playsByDef[p.defId] ?? 0) + 1; plays++ } for (const [k, v] of Object.entries(r.seenDef)) seenByDef[k] = (seenByDef[k] ?? 0) + v; seen += r.seen }
  const perDef = Object.keys(seenByDef).map((d) => ({ id: d, name: getCardDef(d as any).name, sem: primarySemantic(getCardDef(d as any).effects) ?? 'STRIKE', playsPerGame: (playsByDef[d] ?? 0) / n, seenPerGame: seenByDef[d] / n, useWhenSeen: ratio(playsByDef[d] ?? 0, seenByDef[d]) })).sort((a, b) => b.playsPerGame - a.playsPerGame)
  const playVec = perDef.map((d) => d.playsPerGame)
  const semPlays: Record<string, number> = {}, semAp: Record<string, number> = {}
  let apSpent = 0
  for (const r of recs) { for (const p of r.played) semPlays[p.sem] = (semPlays[p.sem] ?? 0) + 1; for (const [k, v] of Object.entries(r.apBySem)) { semAp[k] = (semAp[k] ?? 0) + v; apSpent += v } }
  const plans = new Set(recs.map((r) => r.planSig)), ultSigs = new Set(recs.filter((r) => r.ultSig).map((r) => r.ultSig))
  const ultN = recs.filter((r) => r.ultSig).length
  const ans: Record<string, number> = {}
  for (const r of recs) if (r.ultAnswer) ans[r.ultAnswer] = (ans[r.ultAnswer] ?? 0) + 1
  const ansClass = { guard: 0, oracle: 0, weaken: 0, mend: 0, kill: 0, tank: 0, none: 0 }
  for (const r of recs) if (r.ultAnswer) { if (r.ultAnswer.includes('GUARD')) ansClass.guard++; if (r.ultAnswer.includes('ORACLE')) ansClass.oracle++; if (r.ultAnswer.includes('WEAKEN')) ansClass.weaken++; if (r.ultAnswer.includes('MEND')) ansClass.mend++; if (r.ultAnswer.includes('KILL')) ansClass.kill++; if (r.ultAnswer.includes('TANK')) ansClass.tank++; if (r.ultAnswer.includes('NONE')) ansClass.none++ }
  const ug = recs.filter((r) => r.ultHandGuard === true), ung = recs.filter((r) => r.ultHandGuard === false)
  const preSets = new Set(recs.filter((r) => r.preUltSems).map((r) => r.preUltSems))
  return {
    n, win: won.length / n, lost: lost.length / n, unfinished: fin.length / n, deathClass: Object.fromEntries(Object.entries(cls).map(([k, v]) => [k, v / n])),
    winRoundMean: mean(won.map((r) => r.round)), scoreWonMean: mean(won.map((r) => r.score)), hpWonMean: mean(won.map((r) => r.hp)), winRoundStd: std(won.map((r) => r.round)), reachR6: mean(recs.map((r) => +(r.roundsPlayed >= 5 && (r.status !== 'won' || r.round >= 6)))),
    minHpMean: mean(recs.map((r) => r.minHp)), minHpStd: std(recs.map((r) => r.minHp)), minHpLe50: mean(recs.map((r) => +(r.minHp <= 15))), dmgTakenMean: mean(recs.map((r) => r.dmgTaken.reduce((p, q) => p + q, 0))),
    oracleUses: uses.length / n, shareGuard: ratio(byT[0], uses.length), shareGuide: ratio(byT[1], uses.length), shareStrike: ratio(byT[2], uses.length), guardOnUlt: ratio(uses.filter((u) => u.t === 0 && u.ult).length, byT[0]),
    godStrikeRate: mean(recs.map((r) => +(r.bursts > 0))), ultReach: mean(recs.map((r) => +(r.ultR > 0))), ultGuardRate: ratio(recs.reduce((s, r) => s + r.ultGuarded, 0), recs.reduce((s, r) => s + r.ultR, 0)),
    cardsPlayedMean: plays / n, deadCardRate: 1 - ratio(plays, seen), distinctDefsUsed: perDef.filter((d) => d.playsPerGame >= 0.1).length, deckDistinct: perDef.length, top5Share: ratio(playVec.slice(0, 5).reduce((p, q) => p + q, 0), playVec.reduce((p, q) => p + q, 0)), giniPlays: gini(playVec), perDef,
    semPlayShare: Object.fromEntries(SEMS.map((s) => [s, ratio(semPlays[s] ?? 0, plays)])), semApShare: Object.fromEntries(SEMS.map((s) => [s, ratio(semAp[s] ?? 0, apSpent)])),
    planDistinct: plans.size / n, ultSigDistinct: ratio(ultSigs.size, ultN), ultAnswers: Object.fromEntries(Object.entries(ans).sort((a, b) => b[1] - a[1]).slice(0, 12).map(([k, v]) => [k, v / ultN])), ultAnswerClass: Object.fromEntries(Object.entries(ansClass).map(([k, v]) => [k, ratio(v, ultN)])),
    preUltWeakenRate: mean(recs.map((r) => +r.preUltWeaken)), preUltSetDistinct: preSets.size,
    ultHandGuard: { n: ug.length, win: mean(ug.map((r) => +(r.status === 'won'))), ultDmg: mean(ug.map((r) => r.ultDmg ?? 0)) }, ultHandNoGuard: { n: ung.length, win: mean(ung.map((r) => +(r.status === 'won'))), ultDmg: mean(ung.map((r) => r.ultDmg ?? 0)) },
  }
}
type Agg = ReturnType<typeof aggregate>
const LEVELS: { key: string; stake: Stake }[] = [
  { key: '1', stake: { level: 1 } }, { key: '2', stake: { level: 2 } }, { key: '3', stake: { level: 3 } }, { key: '4', stake: { level: 4 } }, { key: '5', stake: { level: 5 } }, { key: '6', stake: { level: 6 } },
  { key: '7p', stake: { level: 7, choice: 'pressure' } }, { key: '7r', stake: { level: 7, choice: 'race' } }, { key: '7t', stake: { level: 7, choice: 'tempo' } },
]
type Face = { agg: Agg; byEnemy: Record<string, Agg>; byGod: Record<string, Agg>; cells: Record<string, { win: number; lossSeeds: number }>; pairedVsNaive?: Record<string, number> }
type Out = Record<string, Record<string, Record<string, Face>>> // variant → face → policy
function runFace(out: Out, variant: string, faceKey: string, stake: Stake, policies: Policy[], seeds: number, diff: Difficulty, daily: boolean, counter: { games: number }) {
  out[variant][faceKey] = {}
  const perSeedWin: Partial<Record<Policy, Record<string, number[]>>> = {}
  for (const policy of policies) {
    const recs: Rec[] = []; const byEnemy: Record<string, Rec[]> = {}; const byGod: Record<string, Rec[]> = {}; const cells: Face['cells'] = {}
    perSeedWin[policy] = {}
    for (const god of GODS) for (const enemy of ENEMIES) {
      const key = `${god.nameJa}×${enemy.name}`; byEnemy[enemy.name] ??= []; byGod[god.nameJa] ??= []
      const cr: Rec[] = []
      for (let i = 0; i < seeds; i++) { const r = play(god.id, enemy.id, diff, policy, i, stake, daily); recs.push(r); byEnemy[enemy.name].push(r); byGod[god.nameJa].push(r); cr.push(r); counter.games++ }
      perSeedWin[policy]![key] = cr.map((r) => (r.status === 'won' ? 1 : 0))
      cells[key] = { win: mean(cr.map((r) => +(r.status === 'won'))), lossSeeds: cr.filter((r) => r.status !== 'won').length }
    }
    out[variant][faceKey][policy] = { agg: aggregate(recs), byEnemy: Object.fromEntries(Object.entries(byEnemy).map(([k, v]) => [k, aggregate(v)])), byGod: Object.fromEntries(Object.entries(byGod).map(([k, v]) => [k, aggregate(v)])), cells }
  }
  for (const p of ['reader', 'planner'] as Policy[]) if (perSeedWin[p] && perSeedWin.naive) {
    const paired: Record<string, number> = {}
    for (const key of Object.keys(perSeedWin[p]!)) { const a = perSeedWin[p]![key], b = perSeedWin.naive![key]; let div = 0; for (let i = 0; i < a.length; i++) if (a[i] !== b[i]) div++; paired[key] = div / a.length }
    out[variant][faceKey][p].pairedVsNaive = paired
  }
}

const FINAL_POLICIES: Policy[] = [...TUNE_POLICIES, 'oracleAware']
const RANK_POLICIES: Policy[] = ['reader', 'naive', 'greedy', 'planner', 'p_noGuard', 'p_noWeaken', 'p_noMend']
