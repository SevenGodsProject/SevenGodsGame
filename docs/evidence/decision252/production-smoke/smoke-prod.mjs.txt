// 決定252 Production Smoke (scratch). Drives the live UI at process.argv[2] (default Production) and records intents / presentation /
// oracle count / god strike per round. Compares intents with the Pilot After gate (same seeds).
import { chromium } from 'playwright'
import { writeFileSync, mkdirSync, readFileSync } from 'node:fs'
const BASE = process.argv[2] ?? 'https://seven-gods-game.vercel.app'
const OUT = process.argv[3] ?? 'C:/Users/kimi1/SevenGodsGame-d252/docs/evidence/decision252/production-smoke'
mkdirSync(OUT, { recursive: true })
const GOD_INDEX = { ebisu: 0, taiyo: 1 }
const ENEMY_INDEX = { trial: 0, oni: 1, onryo: 2, karakuri: 3, juuma: 4, ryujin: 5, doukeshi: 6 }
const DIFF_INDEX = { easy: 0, normal: 1, hard: 2 }
const CASES = [
  { id: 'taiyo-oni-normal', god: 'taiyo', enemy: 'oni', difficulty: 'normal', stake: 0, seed: 'd252-qa1', rounds: 6, play: true },
  { id: 'taiyo-doukeshi-hard', god: 'taiyo', enemy: 'doukeshi', difficulty: 'hard', stake: 0, seed: 'd252-qa2', rounds: 7, play: true },
  { id: 'ebisu-juuma-stake6', god: 'ebisu', enemy: 'juuma', difficulty: 'normal', stake: 6, seed: 'd252-qa3', rounds: 4, play: false },
  { id: 'taiyo-karakuri-normal', god: 'taiyo', enemy: 'karakuri', difficulty: 'normal', stake: 0, seed: 'd252-qa4', rounds: 5, play: false },
  { id: 'taiyo-onryo-normal', god: 'taiyo', enemy: 'onryo', difficulty: 'normal', stake: 0, seed: 'd252-qa5', rounds: 4, play: false },
  { id: 'taiyo-ryujin-normal', god: 'taiyo', enemy: 'ryujin', difficulty: 'normal', stake: 0, seed: 'd252-qa6', rounds: 4, play: false },
  { id: 'taiyo-trial-normal', god: 'taiyo', enemy: 'trial', difficulty: 'normal', stake: 0, seed: 'd252-qa7', rounds: 5, play: false },
]
const VIEWPORTS = [{ name: 'pc', width: 1508, height: 660 }, { name: 'sp', width: 390, height: 844, mobile: true, only: ['taiyo-oni-normal', 'taiyo-doukeshi-hard'] }]

async function snapshot(page) {
  return page.evaluate(() => {
    const q = (s) => document.querySelector(s)
    const avatar = q('.enemy-avatar'); const after = avatar ? getComputedStyle(avatar, '::after') : null; const intent = q('.intent')
    const box = (el) => { if (!el) return null; const r = el.getBoundingClientRect(); return [Math.round(r.x), Math.round(r.y), Math.round(r.width), Math.round(r.height)] }
    return {
      round: (document.body.innerText.match(/ラウンド\s*(\d)\s*\/\s*7/) || [])[1] ?? null,
      intentText: intent ? intent.textContent.trim() : null, intentClass: intent ? intent.className : null, intentGlyph: intent ? intent.querySelectorAll('svg.intent-glyph').length : 0,
      avatarClass: avatar ? avatar.className : null, avatarBox: box(avatar), avatarScale: avatar ? getComputedStyle(avatar).scale : null,
      footRing: after && after.content !== 'none' ? after.height : null,
      oracle: (document.body.innerText.match(/託宣（残り(\d)回/) || [])[1] ?? null,
      hpText: q('.enemy-plate-status')?.textContent?.trim().slice(0, 40) ?? null,
      hScroll: document.documentElement.scrollWidth > document.documentElement.clientWidth,
      bundle: [...document.scripts].map((s) => (s.src.match(/index-[A-Za-z0-9_-]+\.js/) || [])[0]).filter(Boolean)[0] ?? null,
    }
  })
}

async function runCase(browser, vp, c) {
  const ctx = await browser.newContext({ viewport: { width: vp.width, height: vp.height }, isMobile: !!vp.mobile, hasTouch: !!vp.mobile, deviceScaleFactor: 1 })
  const page = await ctx.newPage()
  const errors = []
  page.on('console', (m) => { if (m.type() === 'error') errors.push(m.text().slice(0, 200)) })
  page.on('pageerror', (e) => errors.push('pageerror ' + String(e).slice(0, 200)))
  const url = `${BASE}/?seed=${c.seed}&enemy=${c.enemy}${c.stake ? `&stake=${c.stake}` : ''}`
  await page.goto(url, { waitUntil: 'networkidle' })
  await page.evaluate(() => { try { localStorage.clear() } catch {} })
  await page.goto(url, { waitUntil: 'networkidle' })
  await page.click('[data-testid="home-start"]')
  await page.locator('.god-select-card').nth(GOD_INDEX[c.god]).click()
  await page.locator('.difficulty-select button').nth(DIFF_INDEX[c.difficulty]).click()
  await page.click('.god-select-confirm')
  if (await page.locator('.enemy-select-card').count()) await page.locator('.enemy-select-card').nth(ENEMY_INDEX[c.enemy]).click()
  await page.getByRole('button', { name: 'この構成でバトル開始' }).click()
  await page.waitForSelector('.enemy-avatar', { timeout: 20000 })
  await page.waitForTimeout(2500)
  const rounds = []
  let godStrike = false, reactions = new Set(), status = null
  for (let r = 1; r <= c.rounds; r++) {
    await page.waitForFunction((rr) => { const m = document.body.innerText.match(/ラウンド\s*(\d)/); return m && m[1] === String(rr) && !document.querySelector('.enemy-cutin') }, r, { timeout: 8000 }).catch(() => {})
    await page.waitForTimeout(400)
    if (!(await page.getByRole('button', { name: 'ラウンドを終える' }).isVisible().catch(() => false))) { status = await page.evaluate(() => document.body.innerText.slice(0, 120).replace(/s+/g, ' ')); break }
    const before = await snapshot(page)
    if (!before.round || Number(before.round) < r) { status = await page.evaluate(() => (document.body.innerText.match(/(勝利|敗北|撃破|未撃破|GAME OVER|VICTORY)[^\n]{0,20}/) || [])[0] ?? null); break }
    if (c.play) {
      // play affordable cards (up to 6 clicks) to exercise reactions and possibly God Strike
      for (let i = 0; i < 4; i++) {
        const cards = page.locator('.card-view:not(.card-view-disabled):not([aria-disabled="true"])')
        if (!(await cards.count())) break
        await cards.first().click().catch(() => {})
        for (let k = 0; k < 6; k++) { await page.waitForTimeout(50); const f = await page.evaluate(() => ({ gs: !!document.querySelector('[class*="god-strike"], .burst-banner, .resonance-cutin, [class*="burst"]'), rl: [...document.querySelectorAll('[class*="rl-"]')].flatMap((e) => [...e.classList].filter((x) => x.startsWith('rl-'))) })); if (f.gs) godStrike = true; f.rl.forEach((x) => reactions.add(x)) }
        await page.waitForTimeout(500)
      }
    }
    const endBtn = page.getByRole('button', { name: 'ラウンドを終える' })
    await endBtn.click({ timeout: 10000 }).catch(() => {})
    let cutinText = null, cutinSeen = false
    for (let i = 0; i < 40; i++) { await page.waitForTimeout(60); const d = await page.evaluate(() => ({ c: !!document.querySelector('.enemy-cutin'), t: document.querySelector('.enemy-cutin-text')?.textContent ?? null, gs: !!document.querySelector('[class*="god-strike"], .burst-banner') })); if (d.c) cutinSeen = true; if (d.t) cutinText = d.t; if (d.gs) godStrike = true }
    rounds.push({ round: r, ...before, cutinSeen, cutinText })
    await page.waitForTimeout(1500)
  }
  const finalHp = await page.evaluate(() => document.body.innerText.match(/敵[^\n]{0,30}/)?.[0] ?? null)
  await page.screenshot({ path: `${OUT}/${c.id}-${vp.name}.png` }).catch(() => {})
  await ctx.close()
  return { rounds, errors, godStrike, reactions: [...reactions], status, finalHp }
}

const browser = await chromium.launch()
const result = { base: BASE, at: new Date().toISOString(), cases: {} }
for (const vp of VIEWPORTS) for (const c of CASES) { if (vp.only && !vp.only.includes(c.id)) continue
  const key = `${c.id}/${vp.name}`
  try { result.cases[key] = await runCase(browser, vp, c); console.log('done', key, 'errors', result.cases[key].errors.length, 'gs', result.cases[key].godStrike) } catch (e) { result.cases[key] = { error: String(e).slice(0, 300) }; console.log('FAIL', key, String(e).slice(0, 200)) }
}
await browser.close()
// parity with Pilot After gate (intent text per round, same seeds)
try {
  const pilot = JSON.parse(readFileSync('C:/Users/kimi1/SevenGodsGame-d252/docs/evidence/decision252/pilot/presentation/gate-presentation.json', 'utf8'))
  const cmp = []
  for (const [k, v] of Object.entries(result.cases)) { const [id, vp] = k.split('/'); const p = pilot[`${id}/${vp}/after`]; if (!p || !v.rounds) continue; for (const r of v.rounds) { const pr = p.rounds.find((x) => x.round === r.round); if (pr) cmp.push({ key: k, round: r.round, prod: r.intentText, pilot: pr.before.intentText, same: r.intentText === pr.before.intentText }) } }
  result.parity = cmp
  console.log('parity mismatches', cmp.filter((x) => !x.same).length, '/', cmp.length)
} catch (e) { console.log('parity skipped', String(e).slice(0, 100)) }
writeFileSync(`${OUT}/smoke252.json`, JSON.stringify(result, null, 1))
console.log('written')
